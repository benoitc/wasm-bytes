<script>
(function () {
  "use strict";

  const TRACK_MAX = 640;   // x runs 0..640
  const SPEED = 110;       // ball speed, pixels per second
  const BEEP_MS = 200;     // beep interval
  const BEEP_LEN = 0.09;   // beep length in seconds
  const BEEP_FREQ = 880;   // beep pitch in Hz

  const $ = (id) => document.getElementById(id);
  const ball = $("ball");
  const reading = $("reading");
  const soundBtn = $("sound");
  const pauseBtn = $("pause");
  const origBtn = $("mode-original");
  const revBtn = $("mode-reversed");

  const screenPan = {};     // key -> exported screen_pan function
  let mode = "original";
  let pos = 0;              // ball position as a float; x = floor(pos)
  let lastT = null;
  let paused = false;
  let soundOn = false;
  let ctx = null;
  let beepTimer = null;

  function b64ToBytes(s) {
    const bin = atob(s);
    const out = new Uint8Array(bin.length);
    for (let i = 0; i < bin.length; i++) out[i] = bin.charCodeAt(i);
    return out;
  }

  async function loadModules() {
    if (!window.PAN_MODULES) throw new Error("modules.js did not load");
    for (const key of ["original", "reversed"]) {
      const def = window.PAN_MODULES[key];
      if (!def || !def.bytes) throw new Error("missing module: " + key);
      const result = await WebAssembly.instantiate(b64ToBytes(def.bytes), {});
      const fn = result.instance.exports.screen_pan;
      if (typeof fn !== "function") throw new Error("screen_pan not exported by " + key);
      screenPan[key] = fn;
    }
  }

  function currentX() {
    return Math.floor(pos);
  }

  // Calls the selected module's screen_pan. Returns null if not ready.
  function panFor(x) {
    const fn = screenPan[mode];
    return fn ? fn(x) : null;
  }

  function render() {
    const x = currentX();
    ball.style.left = "calc(10px + (100% - 20px) * " + (x / TRACK_MAX) + ")";
    const v = panFor(x);
    if (v === null) {
      reading.textContent = "loading…";
    } else {
      reading.textContent = "x = " + x + "    screen_pan(x) = " + v;
    }
  }

  function frame(t) {
    if (lastT === null) lastT = t;
    const dt = (t - lastT) / 1000;
    lastT = t;
    if (!paused) {
      pos = (pos + dt * SPEED) % (TRACK_MAX + 1);
    }
    render();
    requestAnimationFrame(frame);
  }

  function beep(v) {
    if (!ctx) return;
    const now = ctx.currentTime;
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    const panner = ctx.createStereoPanner();

    osc.type = "sine";
    osc.frequency.value = BEEP_FREQ;

    gain.gain.setValueAtTime(0, now);
    gain.gain.linearRampToValueAtTime(0.25, now + 0.005);
    gain.gain.setValueAtTime(0.25, now + BEEP_LEN - 0.015);
    gain.gain.linearRampToValueAtTime(0, now + BEEP_LEN);

    panner.pan.value = v / 500;

    osc.connect(gain);
    gain.connect(panner);
    panner.connect(ctx.destination);

    osc.start(now);
    osc.stop(now + BEEP_LEN + 0.01);
    osc.onended = () => {
      osc.disconnect();
      gain.disconnect();
      panner.disconnect();
    };
  }

  function tick() {
    const v = panFor(currentX());
    if (v !== null) beep(v);
  }

  function setPressed(btn, on) {
    btn.setAttribute("aria-pressed", on ? "true" : "false");
  }

  soundBtn.addEventListener("click", () => {
    if (!soundOn) {
      // Must be created and resumed inside a user gesture for iOS Safari.
      if (!ctx) {
        const AC = window.AudioContext || window.webkitAudioContext;
        ctx = new AC();
      }
      // Lets audio play even when the iPhone's silent switch is on.
      if (navigator.audioSession) {
        try { navigator.audioSession.type = "playback"; } catch (e) { /* ignore */ }
      }
      if (ctx.state !== "running") ctx.resume();

      soundOn = true;
      soundBtn.textContent = "Sound on";
      setPressed(soundBtn, true);
      tick();
      beepTimer = setInterval(tick, BEEP_MS);
    } else {
      soundOn = false;
      clearInterval(beepTimer);
      beepTimer = null;
      soundBtn.textContent = "Sound off";
      setPressed(soundBtn, false);
    }
  });

  pauseBtn.addEventListener("click", () => {
    paused = !paused;
    pauseBtn.textContent = paused ? "Resume" : "Pause";
    setPressed(pauseBtn, paused);
    render();
  });

  function selectMode(key) {
    mode = key;
    setPressed(origBtn, key === "original");
    setPressed(revBtn, key === "reversed");
    render();
  }

  origBtn.addEventListener("click", () => selectMode("original"));
  revBtn.addEventListener("click", () => selectMode("reversed"));

  loadModules()
    .then(() => {
      render();
    })
    .catch((err) => {
      reading.textContent = "Error: " + err.message;
      soundBtn.disabled = true;
    });

  requestAnimationFrame(frame);
})();
</script>
