<script>
(function () {
  'use strict';

  var MAX_X = 640;
  var SPEED = 200; /* units per second */

  var ball = document.getElementById('ball');
  var reading = document.getElementById('reading');
  var note = document.getElementById('note');
  var soundBtn = document.getElementById('sound');
  var originalBtn = document.getElementById('mode-original');
  var reversedBtn = document.getElementById('mode-reversed');
  var pauseBtn = document.getElementById('pause');

  var modules = { original: null, reversed: null };
  var loadNote = null;

  function loadModules() {
    var defs = window.PAN_MODULES;
    if (!defs) {
      loadNote = 'modules.js did not define window.PAN_MODULES';
      return;
    }
    ['original', 'reversed'].forEach(function (key) {
      var entry = defs[key];
      if (!entry || typeof entry.bytes !== 'string') {
        loadNote = key + ': missing base64 bytes';
        return;
      }
      try {
        var b64 = entry.bytes.replace(/\s+/g, '');
        var bin = atob(b64);
        var bytes = new Uint8Array(bin.length);
        for (var i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
        var mod = new WebAssembly.Module(bytes);
        var instance = new WebAssembly.Instance(mod, {});
        var fn = instance.exports && instance.exports.screen_pan;
        if (typeof fn !== 'function') throw new Error('missing screen_pan export');
        modules[key] = {
          label: (entry.label && typeof entry.label === 'string') ? entry.label : key,
          screen_pan: fn
        };
      } catch (err) {
        loadNote = key + ': ' + (err && err.message ? err.message : String(err));
      }
    });
  }

  loadModules();

  if (loadNote) note.textContent = 'Could not load modules.js — ' + loadNote + '.';

  var mode = 'original';
  var soundOn = false;
  var paused = false;
  var pos = 0;
  var x = 0;
  var audioCtx = null;

  function selectedModule() { return modules[mode] || null; }

  function screenPan(xv) {
    var mod = selectedModule();
    if (!mod) return null;
    var r;
    try { r = mod.screen_pan(xv); } catch (e) { return null; }
    if (typeof r !== 'number' || !isFinite(r)) return null;
    return r;
  }

  function ensureAudio() {
    if (!audioCtx) {
      var AC = window.AudioContext || window.webkitAudioContext;
      if (!AC) return null;
      try { audioCtx = new AC(); } catch (e) { return null; }
    }
    if (audioCtx.state === 'suspended') {
      try {
        var p = audioCtx.resume();
        if (p && typeof p.catch === 'function') p.catch(function () {});
      } catch (e) {}
    }
    return audioCtx;
  }

  function playBeep(v) {
    if (!audioCtx) return;
    var ctx = audioCtx;
    var t = ctx.currentTime;
    var dur = 0.1;
    var osc = ctx.createOscillator();
    var gain = ctx.createGain();
    var panner = ctx.createStereoPanner();
    osc.type = 'square';
    osc.frequency.setValueAtTime(880, t);
    panner.pan.setValueAtTime(v / 500, t);
    gain.gain.setValueAtTime(0.0001, t);
    gain.gain.exponentialRampToValueAtTime(0.2, t + 0.01);
    gain.gain.exponentialRampToValueAtTime(0.0001, t + dur - 0.01);
    osc.connect(gain);
    gain.connect(panner);
    panner.connect(ctx.destination);
    osc.start(t);
    osc.stop(t + dur);
    osc.onended = function () {
      try { osc.disconnect(); gain.disconnect(); panner.disconnect(); } catch (e) {}
    };
  }

  function renderReading() {
    var mod = selectedModule();
    var lines = [];
    lines.push(mod ? mod.label : 'unavailable');
    lines.push('x: ' + x);
    var v = mod ? screenPan(x) : null;
    lines.push('screen_pan: ' + (typeof v === 'number' ? String(v) : '—'));
    reading.textContent = lines.join('\n');
  }

  function setMode(m) {
    mode = m;
    originalBtn.setAttribute('aria-pressed', m === 'original' ? 'true' : 'false');
    reversedBtn.setAttribute('aria-pressed', m === 'reversed' ? 'true' : 'false');
    renderReading();
  }

  soundBtn.addEventListener('click', function () {
    soundOn = !soundOn;
    soundBtn.setAttribute('aria-pressed', soundOn ? 'true' : 'false');
    soundBtn.textContent = soundOn ? 'Sound: on' : 'Sound: off';
    if (soundOn) {
      ensureAudio();
      var v = screenPan(x);
      if (typeof v === 'number') playBeep(v);
    }
  });

  originalBtn.addEventListener('click', function () { setMode('original'); });
  reversedBtn.addEventListener('click', function () { setMode('reversed'); });

  pauseBtn.addEventListener('click', function () {
    paused = !paused;
    pauseBtn.setAttribute('aria-pressed', paused ? 'true' : 'false');
    pauseBtn.textContent = paused ? 'Resume' : 'Pause';
  });

  setInterval(function () {
    if (!soundOn) return;
    var v = screenPan(x);
    if (typeof v === 'number') playBeep(v);
  }, 200);

  var lastTs = performance.now();
  function frame(ts) {
    var dt = (ts - lastTs) / 1000;
    lastTs = ts;
    if (dt > 0.1) dt = 0.1;
    if (dt < 0) dt = 0;
    if (!paused) {
      pos += SPEED * dt;
      if (pos >= MAX_X) pos = 0;
    }
    x = Math.round(pos);
    if (x < 0) x = 0;
    if (x > MAX_X) x = MAX_X;
    ball.style.left = (pos / MAX_X * 100) + '%';
    renderReading();
    requestAnimationFrame(frame);
  }

  renderReading();
  requestAnimationFrame(frame);
})();
</script>
