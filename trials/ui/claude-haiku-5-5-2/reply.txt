<script>
(function () {
  'use strict';

  const $ = (id) => document.getElementById(id);
  const soundBtn = $('sound');
  const origBtn = $('mode-original');
  const revBtn = $('mode-reversed');
  const pauseBtn = $('pause');
  const reading = $('reading');
  const ball = $('ball');
  const stage = $('stage');
  const marker = $('marker');
  const rowsEl = $('rows');

  const SPEED = 120;      // stage pixels per second
  const BEEP_EVERY = 200; // ms

  const mods = {};        // key -> { bytes, fn }
  let ready = false;
  let mode = 'original';
  let pos = 0;            // float position, 0..640
  let x = 0;              // integer position, 0..640
  let v = 0;              // last value returned by screen_pan
  let paused = false;
  let last = performance.now();

  let ac = null;
  let soundOn = false;
  let timer = null;

  function pan(xVal) {
    return Number(mods[mode].fn(xVal));
  }

  function setMode(m) {
    mode = m;
    origBtn.setAttribute('aria-pressed', String(m === 'original'));
    revBtn.setAttribute('aria-pressed', String(m === 'reversed'));
    if (ready) v = pan(x);
  }
  origBtn.addEventListener('click', () => setMode('original'));
  revBtn.addEventListener('click', () => setMode('reversed'));

  pauseBtn.addEventListener('click', () => {
    paused = !paused;
    pauseBtn.textContent = paused ? 'Play' : 'Pause';
  });

  soundBtn.addEventListener('click', () => {
    if (!soundOn) {
      if (!ac) {
        const Ctx = window.AudioContext || window.webkitAudioContext;
        if (!Ctx) {
          reading.textContent = 'Web Audio is not available in this browser.';
          return;
        }
        ac = new Ctx();
      }
      // Let playback sound even when the iPhone silent switch is on (Safari 16.4+).
      try {
        if (navigator.audioSession) navigator.audioSession.type = 'playback';
      } catch (e) { /* ignore */ }
      if (ac.state !== 'running') ac.resume();
      soundOn = true;
      soundBtn.textContent = 'Sound: on';
      beep();
      timer = setInterval(beep, BEEP_EVERY);
    } else {
      soundOn = false;
      clearInterval(timer);
      timer = null;
      soundBtn.textContent = 'Sound: off';
    }
  });

  function beep() {
    if (!ac || !ready) return;
    const t = ac.currentTime;
    const osc = ac.createOscillator();
    const gain = ac.createGain();
    const panner = ac.createStereoPanner();
    osc.type = 'sine';
    osc.frequency.value = 880;
    gain.gain.setValueAtTime(0, t);
    gain.gain.linearRampToValueAtTime(0.25, t + 0.005);
    gain.gain.setValueAtTime(0.25, t + 0.055);
    gain.gain.linearRampToValueAtTime(0, t + 0.07);
    panner.pan.value = pan(x) / 500;
    osc.connect(gain);
    gain.connect(panner);
    panner.connect(ac.destination);
    osc.start(t);
    osc.stop(t + 0.08);
  }

  function render() {
    const W = stage.clientWidth;
    const center = 16 + (x / 640) * (W - 32);
    ball.style.left = (center - 11) + 'px';

    reading.textContent = 'x = ' + x + '\nscreen_pan(x) = ' + v;
    const half = reading.offsetWidth / 2;
    const clamped = Math.min(Math.max(center, half + 4), W - half - 4);
    reading.style.left = clamped + 'px';

    marker.style.left = (50 + v / 10) + '%';
  }

  function frame(now) {
    const dt = Math.min(0.1, (now - last) / 1000);
    last = now;
    if (!paused) pos = (pos + dt * SPEED) % 640;
    x = Math.round(pos);
    if (ready) v = pan(x);
    render();
    requestAnimationFrame(frame);
  }

  async function loadModule(key) {
    const info = window.PAN_MODULES[key];
    const bin = atob(info.bytes);
    const bytes = new Uint8Array(bin.length);
    for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
    const result = await WebAssembly.instantiate(bytes, {});
    const fn = result.instance.exports.screen_pan;
    if (typeof fn !== 'function') throw new Error(key + ' does not export screen_pan');
    return { bytes, fn };
  }

  function hex(buf) {
    return Array.from(new Uint8Array(buf), (b) => b.toString(16).padStart(2, '0')).join('');
  }

  function buildRows() {
    const cells = {};
    for (const key of ['original', 'reversed']) {
      const info = window.PAN_MODULES[key];
      const tr = document.createElement('tr');

      const tdLabel = document.createElement('td');
      tdLabel.textContent = info.label;

      const tdModel = document.createElement('td');
      const model = document.createElement('div');
      model.className = 'model';
      model.textContent = info.model;
      const reply = document.createElement('div');
      reply.className = 'muted small';
      reply.textContent = info.reply;
      tdModel.append(model, reply);

      const tdRecorded = document.createElement('td');
      const rec = document.createElement('div');
      rec.className = 'hash';
      rec.textContent = info.sha256;
      tdRecorded.appendChild(rec);

      const tdLoaded = document.createElement('td');
      tdLoaded.textContent = 'Computing...';
      cells[key] = tdLoaded;

      tr.append(tdLabel, tdModel, tdRecorded, tdLoaded);
      rowsEl.appendChild(tr);
    }
    return cells;
  }

  async function fillHashCell(key, cell) {
    const recorded = String(window.PAN_MODULES[key].sha256 || '').trim().toLowerCase();
    const m = mods[key];
    if (!m) {
      cell.textContent = 'Not loaded.';
      return;
    }
    if (!(window.crypto && window.crypto.subtle)) {
      cell.textContent = 'Not computed: SHA-256 is not available in this browser context.';
      return;
    }
    try {
      const h = hex(await window.crypto.subtle.digest('SHA-256', m.bytes));
      const ok = h === recorded;
      cell.textContent = '';
      const code = document.createElement('div');
      code.className = 'hash';
      code.textContent = h;
      const note = document.createElement('div');
      note.className = 'muted small';
      note.textContent = ok
        ? 'Matches the recorded SHA-256.'
        : 'Does not match the recorded SHA-256.';
      cell.append(code, note);
    } catch (e) {
      cell.textContent = 'Not computed: ' + e.message;
    }
  }

  async function init() {
    if (!window.PAN_MODULES) {
      reading.textContent = 'modules.js did not load.';
      requestAnimationFrame(frame);
      return;
    }
    const cells = buildRows();
    try {
      for (const key of ['original', 'reversed']) {
        mods[key] = await loadModule(key);
      }
    } catch (e) {
      reading.textContent = 'Could not load the modules: ' + e.message;
      requestAnimationFrame(frame);
      return;
    }
    ready = true;
    v = pan(x);
    setMode(mode);
    for (const key of ['original', 'reversed']) {
      fillHashCell(key, cells[key]);
    }
    requestAnimationFrame(frame);
  }

  window.addEventListener('resize', render);
  init();
})();
</script>
