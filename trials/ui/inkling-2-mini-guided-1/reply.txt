<script>
(function () {
  'use strict';

  var TRAVEL_MS = 6000;   /* one full left-to-right pass */
  var X_MAX = 640;
  var BEEP_MS = 200;
  var BEEP_LENGTH = 0.09;
  var BEEP_FREQ = 660;
  var BEEP_VOLUME = 0.22;
  var BALL_SIZE = 28;

  var els = {
    sound: document.getElementById('sound'),
    modeOriginal: document.getElementById('mode-original'),
    modeReversed: document.getElementById('mode-reversed'),
    pause: document.getElementById('pause'),
    reading: document.getElementById('reading'),
    track: document.getElementById('track'),
    ball: document.getElementById('ball')
  };

  var state = {
    x: 0,
    elapsed: 0,
    last: null,
    paused: false,
    soundOn: false,
    selected: 'original',
    ready: false
  };

  var modules = { original: null, reversed: null };
  var labels = { original: 'Original', reversed: 'Reversed' };
  var audioCtx = null;
  var beepTimer = null;
  var active = [];

  function base64ToBytes(b64) {
    var clean = String(b64 || '').replace(/\s+/g, '');
    var bin = atob(clean);
    var bytes = new Uint8Array(bin.length);
    for (var i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
    return bytes;
  }

  function loadModule(spec, name) {
    if (!spec || typeof spec.bytes !== 'string') {
      return Promise.reject(new Error(name + ': missing bytes'));
    }
    var bytes = base64ToBytes(spec.bytes);
    return WebAssembly.instantiate(bytes, {}).then(function (result) {
      var fn = result.instance.exports.screen_pan;
      if (typeof fn !== 'function') {
        throw new Error(name + ': module does not export screen_pan');
      }
      return fn;
    });
  }

  function init() {
    var specs = window.PAN_MODULES;
    if (!specs) {
      setReading('modules.js not found');
      return;
    }
    labels.original = (specs.original && specs.original.label) || 'Original';
    labels.reversed = (specs.reversed && specs.reversed.label) || 'Reversed';
    Promise.all([
      loadModule(specs.original, 'original'),
      loadModule(specs.reversed, 'reversed')
    ]).then(function (fns) {
      modules.original = fns[0];
      modules.reversed = fns[1];
      state.ready = true;
      updateReading();
    }).catch(function (err) {
      setReading('failed to load modules: ' + (err && err.message ? err.message : err));
    });
  }

  function selectedFn() {
    return modules[state.selected] || null;
  }

  function currentLabel() {
    return labels[state.selected] || state.selected;
  }

  function setReading(text) {
    els.reading.textContent = text;
  }

  function updateReading() {
    var fn = selectedFn();
    if (!fn) {
      setReading('loading modules…');
      return;
    }
    var v;
    try {
      v = fn(state.x);
    } catch (err) {
      setReading(currentLabel() + ' · x = ' + state.x + ' · screen_pan(x) = error');
      return;
    }
    els.reading.innerHTML = '';
    var mode = document.createElement('span');
    mode.className = 'mode';
    mode.textContent = currentLabel() + '  ';
    var rest = document.createTextNode('x = ' + state.x + '  ·  screen_pan(x) = ');
    var val = document.createElement('span');
    val.className = 'val';
    val.textContent = String(v);
    els.reading.appendChild(mode);
    els.reading.appendChild(rest);
    els.reading.appendChild(val);
  }

  function render() {
    var pct = state.x / X_MAX;
    els.ball.style.left = 'calc(' + (BALL_SIZE / 2) + 'px + (100% - ' + BALL_SIZE + 'px) * ' + pct + ')';
  }

  function frame(now) {
    if (state.last === null) state.last = now;
    var dt = now - state.last;
    state.last = now;
    if (!state.paused) {
      state.elapsed += dt;
      var t = (state.elapsed % TRAVEL_MS) / TRAVEL_MS;
      state.x = Math.min(X_MAX, Math.floor(t * (X_MAX + 1)));
    }
    render();
    updateReading();
    requestAnimationFrame(frame);
  }

  function stopActiveBeeps() {
    for (var i = 0; i < active.length; i++) {
      try { active[i].stop(); } catch (e) {}
    }
    active = [];
  }

  function playBeep() {
    if (!state.soundOn) return;
    if (!audioCtx) return;
    var fn = selectedFn();
    if (!fn) return;
    var v;
    try {
      v = fn(state.x);
    } catch (err) {
      return;
    }
    var pan = v / 500;
    var t = audioCtx.currentTime;
    var osc = audioCtx.createOscillator();
    var gain = audioCtx.createGain();
    var panner = audioCtx.createStereoPanner();
    osc.type = 'square';
    osc.frequency.setValueAtTime(BEEP_FREQ, t);
    gain.gain.setValueAtTime(0.0001, t);
    gain.gain.exponentialRampToValueAtTime(BEEP_VOLUME, t + 0.005);
    gain.gain.exponentialRampToValueAtTime(0.0001, t + BEEP_LENGTH);
    panner.pan.setValueAtTime(pan, t);
    osc.connect(gain);
    gain.connect(panner);
    panner.connect(audioCtx.destination);
    osc.start(t);
    osc.stop(t + BEEP_LENGTH + 0.02);
    active.push(osc);
    osc.onended = function () {
      var idx = active.indexOf(osc);
      if (idx >= 0) active.splice(idx, 1);
    };
  }

  function startSound() {
    if (navigator.audioSession) {
      try { navigator.audioSession.type = 'playback'; } catch (e) {}
    }
    var Ctx = window.AudioContext || window.webkitAudioContext;
    if (!audioCtx && Ctx) {
      try { audioCtx = new Ctx(); } catch (e) { audioCtx = null; }
    }
    if (audioCtx && audioCtx.state === 'suspended') {
      try {
        var p = audioCtx.resume();
        if (p && typeof p.catch === 'function') p.catch(function () {});
      } catch (e) {}
    }
    state.soundOn = true;
    els.sound.textContent = 'Sound: on';
    els.sound.setAttribute('aria-pressed', 'true');
    playBeep();
    beepTimer = setInterval(playBeep, BEEP_MS);
  }

  function stopSound() {
    state.soundOn = false;
    if (beepTimer) { clearInterval(beepTimer); beepTimer = null; }
    stopActiveBeeps();
    els.sound.textContent = 'Sound: off';
    els.sound.setAttribute('aria-pressed', 'false');
  }

  function selectMode(mode) {
    state.selected = mode;
    var orig = mode === 'original';
    els.modeOriginal.setAttribute('aria-pressed', String(orig));
    els.modeReversed.setAttribute('aria-pressed', String(!orig));
    updateReading();
  }

  els.sound.addEventListener('click', function () {
    if (state.soundOn) {
      stopSound();
    } else {
      startSound();
    }
  });

  els.modeOriginal.addEventListener('click', function () { selectMode('original'); });
  els.modeReversed.addEventListener('click', function () { selectMode('reversed'); });

  els.pause.addEventListener('click', function () {
    state.paused = !state.paused;
    els.pause.textContent = state.paused ? 'Resume' : 'Pause';
    els.pause.setAttribute('aria-pressed', String(state.paused));
  });

  init();
  selectMode('original');
  requestAnimationFrame(frame);
})();
</script>
