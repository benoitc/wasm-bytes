<script>
(function(){
  var MAXX = 640, SWEEP_MS = 3200;
  var fns = {}, mode = "original", paused = false, soundOn = false;
  var x = 0, pos = 0, last = null;
  var ctx = null, unlocked = false, beepTimer = null;
  var $ = function(id){return document.getElementById(id)};
  var ball = $("ball"), track = $("track"), reading = $("reading"), status = $("status");

  function b64(s){
    var bin = atob(s), a = new Uint8Array(bin.length);
    for (var i=0;i<bin.length;i++) a[i]=bin.charCodeAt(i);
    return a;
  }

  function load(){
    var m = window.PAN_MODULES;
    if (!m) { status.textContent = "modules.js not loaded."; return Promise.resolve(); }
    var keys = ["original","reversed"];
    return Promise.all(keys.map(function(k){
      if (!m[k]) return;
      return WebAssembly.instantiate(b64(m[k].bytes), {}).then(function(r){
        fns[k] = r.instance.exports.screen_pan;
      });
    })).then(function(){ update(); }).catch(function(e){
      status.textContent = "Could not load module: " + e.message;
    });
  }

  function pan(){
    var f = fns[mode];
    return f ? f(x) : null;
  }

  function update(){
    var v = pan();
    reading.textContent = "x = " + x + ", screen_pan(x) = " + (v === null ? "…" : v);
    var w = track.clientWidth - 26;
    ball.style.transform = "translateX(" + (x / MAXX * w) + "px)";
  }

  function frame(t){
    if (last === null) last = t;
    var dt = t - last; last = t;
    if (!paused) {
      pos += dt / SWEEP_MS * MAXX;
      if (pos > MAXX) pos = pos % (MAXX + 1);
      x = Math.max(0, Math.min(MAXX, Math.floor(pos)));
    }
    update();
    requestAnimationFrame(frame);
  }

  // iOS: route Web Audio to playback channel so silent switch doesn't mute it
  function setPlaybackSession(){
    try { if (navigator.audioSession) navigator.audioSession.type = "playback"; } catch(e){}
  }
  var silentEl = null;
  function keepMediaAlive(){
    // Playing a media element also moves iOS audio to the playback category.
    try {
      if (!silentEl) {
        var n = 4410, buf = new ArrayBuffer(44 + n*2), d = new DataView(buf);
        var w = function(o,s){for(var i=0;i<s.length;i++)d.setUint8(o+i,s.charCodeAt(i))};
        w(0,"RIFF"); d.setUint32(4,36+n*2,true); w(8,"WAVEfmt ");
        d.setUint32(16,16,true); d.setUint16(20,1,true); d.setUint16(22,1,true);
        d.setUint32(24,44100,true); d.setUint32(28,88200,true);
        d.setUint16(32,2,true); d.setUint16(34,16,true);
        w(36,"data"); d.setUint32(40,n*2,true);
        var blob = new Blob([buf],{type:"audio/wav"});
        silentEl = new Audio(URL.createObjectURL(blob));
        silentEl.loop = true;
        silentEl.setAttribute("playsinline","");
      }
      var p = silentEl.play(); if (p && p.catch) p.catch(function(){});
    } catch(e){}
  }

  function ensureCtx(){
    if (!ctx) {
      var AC = window.AudioContext || window.webkitAudioContext;
      if (!AC) { status.textContent = "Web Audio not supported."; return false; }
      ctx = new AC();
    }
    if (ctx.state !== "running") {
      var p = ctx.resume(); if (p && p.catch) p.catch(function(){});
    }
    return true;
  }

  function beep(){
    if (!ctx || !soundOn || ctx.state === "closed") return;
    if (ctx.state !== "running") ctx.resume();
    var v = pan();
    if (v === null) return;
    var t = ctx.currentTime + 0.01;
    var osc = ctx.createOscillator();
    var g = ctx.createGain();
    var p = ctx.createStereoPanner ? ctx.createStereoPanner() : null;
    osc.type = "square";
    osc.frequency.value = 880;
    g.gain.setValueAtTime(0, t);
    g.gain.linearRampToValueAtTime(0.25, t + 0.005);
    g.gain.setValueAtTime(0.25, t + 0.07);
    g.gain.linearRampToValueAtTime(0, t + 0.08);
    osc.connect(g);
    if (p) { p.pan.value = Math.max(-1, Math.min(1, v / 500)); g.connect(p); p.connect(ctx.destination); }
    else g.connect(ctx.destination);
    osc.start(t);
    osc.stop(t + 0.09);
    osc.onended = function(){ try{osc.disconnect();g.disconnect();if(p)p.disconnect();}catch(e){} };
  }

  function setSound(on){
    soundOn = on;
    var b = $("sound");
    b.setAttribute("aria-pressed", on ? "true" : "false");
    b.textContent = "Sound: " + (on ? "on" : "off");
    clearInterval(beepTimer); beepTimer = null;
    if (on) {
      setPlaybackSession();
      keepMediaAlive();
      if (!ensureCtx()) { soundOn = false; return; }
      // unlock with a silent tick inside the gesture
      if (!unlocked) {
        try {
          var s = ctx.createBufferSource();
          s.buffer = ctx.createBuffer(1,1,22050);
          s.connect(ctx.destination); s.start(0);
          unlocked = true;
        } catch(e){}
      }
      beep();
      beepTimer = setInterval(beep, 200);
    } else if (silentEl) {
      try { silentEl.pause(); } catch(e){}
    }
  }

  function setMode(m){
    mode = m;
    $("mode-original").setAttribute("aria-pressed", m === "original");
    $("mode-reversed").setAttribute("aria-pressed", m === "reversed");
    update();
  }

  $("sound").addEventListener("click", function(){ setSound(!soundOn); });
  $("mode-original").addEventListener("click", function(){ setMode("original"); });
  $("mode-reversed").addEventListener("click", function(){ setMode("reversed"); });
  $("pause").addEventListener("click", function(){
    paused = !paused;
    this.setAttribute("aria-pressed", paused ? "true" : "false");
    this.textContent = paused ? "Resume" : "Pause";
  });
  document.addEventListener("visibilitychange", function(){
    if (!document.hidden && soundOn && ctx && ctx.state !== "running") ctx.resume();
  });
  window.addEventListener("resize", update);

  load();
  requestAnimationFrame(frame);
})();
</script>
