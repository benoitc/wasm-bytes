<script>
(function(){
  "use strict";
  var PERIOD = 4000;
  var mods = window.PAN_MODULES || {};
  var fns = {original:null, reversed:null};
  var mode = "original";
  var paused = false;
  var phase = 0, last = null;
  var curX = 0;

  var $ = function(id){return document.getElementById(id)};
  var stage=$("stage"), ball=$("ball"), reading=$("reading"), marker=$("marker");
  var soundBtn=$("sound"), pauseBtn=$("pause");
  var modeBtns={original:$("mode-original"), reversed:$("mode-reversed")};

  function b64bytes(s){
    var bin = atob(String(s||"").replace(/\s+/g,""));
    var a = new Uint8Array(bin.length);
    for(var i=0;i<bin.length;i++) a[i]=bin.charCodeAt(i);
    return a;
  }

  function hex(buf){
    var a=new Uint8Array(buf), s="";
    for(var i=0;i<a.length;i++) s+=(a[i]<16?"0":"")+a[i].toString(16);
    return s;
  }

  // Files table
  var tbody=$("files");
  var checkCells={};
  ["original","reversed"].forEach(function(k){
    var m=mods[k]||{};
    var tr=document.createElement("tr");
    function cell(label,text,cls){
      var td=document.createElement("td");
      td.setAttribute("data-label",label);
      if(cls) td.className=cls;
      td.textContent=text==null?"":String(text);
      tr.appendChild(td);
      return td;
    }
    cell("Module", m.label||k);
    cell("Model", m.model);
    cell("Reply", m.reply);
    cell("Recorded SHA-256", m.sha256, "hash");
    checkCells[k]=cell("Check","checking…");
    tbody.appendChild(tr);
  });

  function verify(k, bytes){
    var td=checkCells[k], m=mods[k]||{};
    if(!bytes){ td.textContent="module not loaded"; return; }
    if(!(window.crypto && crypto.subtle && crypto.subtle.digest)){
      td.textContent="not available in this browser context"; return;
    }
    crypto.subtle.digest("SHA-256", bytes).then(function(d){
      var h=hex(d);
      var ok = h===String(m.sha256||"").toLowerCase().trim();
      td.textContent=(ok?"Matches the recorded value. ":"Does NOT match the recorded value. ")+"Computed: "+h;
    }, function(){ td.textContent="not available in this browser context"; });
  }

  // Load modules
  ["original","reversed"].forEach(function(k){
    var m=mods[k], bytes=null;
    try{
      bytes=b64bytes(m && m.bytes);
    }catch(e){ verify(k,null); return; }
    verify(k,bytes);
    WebAssembly.instantiate(bytes.slice(), {}).then(function(r){
      fns[k]=r.instance.exports.screen_pan;
      render(true);
    }).catch(function(){
      checkCells[k].textContent += " (module failed to instantiate)";
    });
  });

  function pan(x){
    var f=fns[mode];
    return f ? f(x) : null;
  }

  // Rendering
  var lastShown=null;
  function render(force){
    var x=curX, v=pan(x);
    var W=stage.clientWidth;
    var cx=16+(W-32)*x/640;
    ball.style.left=(cx-11)+"px";
    var key=x+"|"+v+"|"+mode+"|"+W;
    if(!force && key===lastShown) return;
    lastShown=key;
    reading.children[0].textContent="x = "+x;
    reading.children[1].textContent="screen_pan(x) = "+(v==null?"…":v);
    var rw=reading.offsetWidth;
    var left=Math.max(0,Math.min(W-rw, cx-rw/2));
    reading.style.left=left+"px";
    if(v!=null){
      marker.style.left=(50+50*v/500)+"%";
    }
  }

  function frame(t){
    if(last==null) last=t;
    if(!paused) phase=(phase+(t-last))%PERIOD;
    last=t;
    curX=Math.min(640, Math.floor(phase/PERIOD*641));
    render(false);
    requestAnimationFrame(frame);
  }
  requestAnimationFrame(frame);
  window.addEventListener("resize", function(){render(true)});

  // Controls
  function setMode(k){
    mode=k;
    for(var n in modeBtns) modeBtns[n].setAttribute("aria-pressed", n===k?"true":"false");
    render(true);
  }
  modeBtns.original.addEventListener("click", function(){setMode("original")});
  modeBtns.reversed.addEventListener("click", function(){setMode("reversed")});

  pauseBtn.addEventListener("click", function(){
    paused=!paused;
    pauseBtn.textContent=paused?"Play":"Pause";
  });

  // Audio
  var ctx=null, timer=null, soundOn=false, streamAudio=null;
  var isIOS=/iPad|iPhone|iPod/.test(navigator.userAgent) ||
            (navigator.platform==="MacIntel" && navigator.maxTouchPoints>1);

  function setupAudio(){
    if(ctx) return;
    try{
      if(navigator.audioSession) navigator.audioSession.type="playback";
    }catch(e){}
    var AC=window.AudioContext||window.webkitAudioContext;
    ctx=new AC();
    ctx.out=ctx.destination;
    if(isIOS && !(navigator.audioSession) && ctx.createMediaStreamDestination){
      try{
        var sd=ctx.createMediaStreamDestination();
        streamAudio=document.createElement("audio");
        streamAudio.setAttribute("playsinline","");
        streamAudio.srcObject=sd.stream;
        ctx.out=sd;
      }catch(e){ ctx.out=ctx.destination; }
    }
  }

  function beep(){
    if(!ctx) return;
    var v=pan(curX);
    if(v==null) return;
    var t=ctx.currentTime+0.01;
    var o=ctx.createOscillator(), g=ctx.createGain();
    o.type="sine"; o.frequency.value=880;
    g.gain.setValueAtTime(0,t);
    g.gain.linearRampToValueAtTime(0.3,t+0.005);
    g.gain.setValueAtTime(0.3,t+0.07);
    g.gain.linearRampToValueAtTime(0,t+0.08);
    var node=g;
    o.connect(g);
    if(ctx.createStereoPanner){
      var p=ctx.createStereoPanner();
      p.pan.value=Math.max(-1,Math.min(1,v/500));
      g.connect(p); node=p;
    }
    node.connect(ctx.out);
    o.start(t); o.stop(t+0.09);
  }

  soundBtn.addEventListener("click", function(){
    soundOn=!soundOn;
    soundBtn.textContent=soundOn?"Sound: on":"Sound: off";
    soundBtn.setAttribute("aria-pressed", soundOn?"true":"false");
    if(soundOn){
      setupAudio();
      if(ctx.state!=="running") ctx.resume();
      if(streamAudio){ var pr=streamAudio.play(); if(pr&&pr.catch) pr.catch(function(){}); }
      beep();
      timer=setInterval(beep,200);
    }else{
      clearInterval(timer); timer=null;
      if(streamAudio) streamAudio.pause();
    }
  });
})();
</script>
