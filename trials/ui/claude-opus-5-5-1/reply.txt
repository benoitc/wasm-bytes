<script>
(function () {
  "use strict";

  var X_MAX = 640;
  var SPEED = 160;          // x units per second
  var BEEP_INTERVAL = 200;  // ms
  var BEEP_FREQ = 880;      // Hz
  var BEEP_LEN = 0.06;      // s
  var BEEP_GAIN = 0.3;

  var el = {
    sound: document.getElementById("sound"),
    orig: document.getElementById("mode-original"),
    rev: document.getElementById("mode-reversed"),
    pause: document.getElementById("pause"),
    reading: document.getElementById("reading"),
    ball: document.getElementById("ball"),
    status: document.getElementById("status"),
    metaLabel: document.getElementById("meta-label"),
    metaSha: document.getElementById("meta-sha")
  };

  var fns = { original: null, reversed: null };
  var info = { original: null, reversed: null };
  var mode = "original";
  var x = 0;
  var pos = 0;              // floating position
  var paused = false;
  var lastT = null;
  var shownX = null, shownMode = null;

  var soundOn = false;
  var ctx = null;
  var beepTimer = null;
  var silentAudio = null;

  function setStatus(msg, isErr) {
    el.status.textContent = msg;
    el.status.className = "status" + (isErr ? " err" : "");
  }

  function base64ToBytes(b64) {
    var clean = String(b64).replace(/[^A-Za-z0-9+/=_-]/g, "").replace(/-/g, "+").replace(/_/g, "/");
    while (clean.length % 4) clean += "=";
    var bin = atob(clean);
    var out = new Uint8Array(bin.length);
    for (var i = 0; i < bin.length; i++) out[i] = bin.charCodeAt(i);
    return out;
  }

  // Provide harmless stubs for any imports a module might declare.
  function buildImports(module) {
    var imports = {};
    var list = WebAssembly.Module.imports(module);
    for (var i = 0; i < list.length; i++) {
      var imp = list[i];
      if (!imports[imp.module]) imports[imp.module] = {};
      var v;
      switch (imp.kind) {
        case "function": v = function () { return 0; }; break;
        case "memory": v = new WebAssembly.Memory({ initial: 1, maximum: 65536 }); break;
        case "table": v = new WebAssembly.Table({ initial: 0, element: "anyfunc" }); break;
        case "global":
          try { v = new WebAssembly.Global({ value: "i32", mutable: false }, 0); } catch (e) { v = 0; }
          break;
        default: v = undefined;
      }
      imports[imp.module][imp.name] = v;
    }
    return imports;
  }

  function loadModule(key) {
    var m = window.PAN_MODULES[key];
    if (!m || typeof m.bytes !== "string") return Promise.reject(new Error("module '" + key + "' missing"));
    var bytes = base64ToBytes(m.bytes);
    return WebAssembly.compile(bytes).then(function (module) {
      return WebAssembly.instantiate(module, buildImports(module));
    }).then(function (instance) {
      var f = instance.exports.screen_pan;
      if (typeof f !== "function") throw new Error("module '" + key + "' does not export screen_pan");
      fns[key] = f;
      info[key] = m;
    });
  }

  function pan(at) {
    var f = fns[mode];
    var v = f(at);
    return Number(v);
  }

  function updateMeta() {
    var m = info[mode];
    el.metaLabel.textContent = m && m.label ? String(m.label) : mode;
    el.metaSha.textContent = m && m.sha256 ? String(m.sha256) : "–";
  }

  function render(force) {
    el.ball.style.setProperty("--x", x);
    el.reading.style.setProperty("--x", x);
    if (!fns[mode]) return;
    if (force || x !== shownX || mode !== shownMode) {
      var v;
      try { v = pan(x); } catch (e) { v = NaN; }
      shownX = x; shownMode = mode;
      el.reading.textContent = "";
      el.reading.appendChild(document.createTextNode("x = " + x + ", screen_pan(x) = "));
      var s = document.createElement("span");
      s.className = "v";
      s.textContent = String(v);
      el.reading.appendChild(s);
    }
  }

  function frame(t) {
    if (lastT === null) lastT = t;
    var dt = Math.min(0.1, (t - lastT) / 1000);
    lastT = t;
    if (!paused) {
      pos += SPEED * dt;
      if (pos > X_MAX) pos = pos % (X_MAX + 1);
      x = Math.max(0, Math.min(X_MAX, Math.round(pos)));
    }
    render(false);
    requestAnimationFrame(frame);
  }

  // ---------- Audio ----------

  function makeSilentWavUrl() {
    var rate = 8000, n = 4000; // 0.5 s mono 16-bit
    var buf = new ArrayBuffer(44 + n * 2);
    var d = new DataView(buf);
    function str(o, s) { for (var i = 0; i < s.length; i++) d.setUint8(o + i, s.charCodeAt(i)); }
    str(0, "RIFF"); d.setUint32(4, 36 + n * 2, true); str(8, "WAVE");
    str(12, "fmt "); d.setUint32(16, 16, true); d.setUint16(20, 1, true); d.setUint16(22, 1, true);
    d.setUint32(24, rate, true); d.setUint32(28, rate * 2, true); d.setUint16(32, 2, true); d.setUint16(34, 16, true);
    str(36, "data"); d.setUint32(40, n * 2, true);
    return URL.createObjectURL(new Blob([buf], { type: "audio/wav" }));
  }

  function unlockAudio() {
    // Ask iOS to treat this page's audio as media playback (ignores the silent switch).
    try {
      if (navigator.audioSession && "type" in navigator.audioSession) navigator.audioSession.type = "playback";
    } catch (e) {}

    if (!ctx) {
      var AC = window.AudioContext || window.webkitAudioContext;
      if (!AC) { setStatus("Web Audio is not supported in this browser.", true); return false; }
      ctx = new AC();
    }

    // Fallback for older iOS: a playing <audio> element moves the session to "playback".
    if (!navigator.audioSession) {
      try {
        if (!silentAudio) {
          silentAudio = document.createElement("audio");
          silentAudio.setAttribute("playsinline", "");
          silentAudio.setAttribute("x-webkit-airplay", "deny");
          silentAudio.loop = true;
          silentAudio.preload = "auto";
          silentAudio.src = makeSilentWavUrl();
        }
        var p = silentAudio.play();
        if (p && p.catch) p.catch(function () {});
      } catch (e) {}
    }

    if (ctx.state !== "running") {
      var r = ctx.resume();
      if (r && r.catch) r.catch(function () {});
    }
    // Play one silent sample inside the gesture (needed by older Safari).
    try {
      var b = ctx.createBuffer(1, 1, ctx.sampleRate);
      var s = ctx.createBufferSource();
      s.buffer = b; s.connect(ctx.destination); s.start(0);
    } catch (e) {}
    return true;
  }

  function beep() {
    if (!soundOn || !ctx || !fns[mode]) return;
    if (ctx.state !== "running") { var r = ctx.resume(); if (r && r.catch) r.catch(function () {}); }
    var v;
    try { v = pan(x); } catch (e) { return; }
    if (!isFinite(v)) return;
    var t = ctx.currentTime + 0.01;
    var osc = ctx.createOscillator();
    var g = ctx.createGain();
    var p = ctx.createStereoPanner ? ctx.createStereoPanner() : null;
    osc.type = "square";
    osc.frequency.setValueAtTime(BEEP_FREQ, t);
    g.gain.setValueAtTime(0, t);
    g.gain.linearRampToValueAtTime(BEEP_GAIN, t + 0.004);
    g.gain.setValueAtTime(BEEP_GAIN, t + BEEP_LEN - 0.01);
    g.gain.linearRampToValueAtTime(0, t + BEEP_LEN);
    osc.connect(g);
    if (p) {
      p.pan.setValueAtTime(v / 500, t);
      g.connect(p); p.connect(ctx.destination);
    } else {
      g.connect(ctx.destination);
    }
    osc.start(t);
    osc.stop(t + BEEP_LEN + 0.02);
    osc.onended = function () {
      try { osc.disconnect(); g.disconnect(); if (p) p.disconnect(); } catch (e) {}
    };
  }

  function setSound(on) {
    if (on) {
      if (!unlockAudio()) return;
      soundOn = true;
      if (beepTimer) clearInterval(beepTimer);
      beep();
      beepTimer = setInterval(beep, BEEP_INTERVAL);
    } else {
      soundOn = false;
      if (beepTimer) { clearInterval(beepTimer); beepTimer = null; }
      if (silentAudio) { try { silentAudio.pause(); } catch (e) {} }
    }
    el.sound.setAttribute("aria-pressed", soundOn ? "true" : "false");
    el.sound.textContent = soundOn ? "Sound: on" : "Sound: off";
  }

  function setMode(m) {
    if (!fns[m]) return;
    mode = m;
    el.orig.setAttribute("aria-pressed", m === "original" ? "true" : "false");
    el.rev.setAttribute("aria-pressed", m === "reversed" ? "true" : "false");
    updateMeta();
    render(true);
  }

  function setPaused(p) {
    paused = p;
    el.pause.setAttribute("aria-pressed", p ? "true" : "false");
    el.pause.textContent = p ? "Resume" : "Pause";
  }

  el.sound.addEventListener("click", function () { setSound(!soundOn); });
  el.orig.addEventListener("click", function () { setMode("original"); });
  el.rev.addEventListener("click", function () { setMode("reversed"); });
  el.pause.addEventListener("click", function () { setPaused(!paused); });

  document.addEventListener("visibilitychange", function () {
    if (!document.hidden && soundOn && ctx && ctx.state !== "running") {
      var r = ctx.resume(); if (r && r.catch) r.catch(function () {});
    }
  });

  // ---------- Start ----------
  if (!window.PAN_MODULES) {
    setStatus("modules.js was not found or did not define PAN_MODULES.", true);
    return;
  }
  if (typeof WebAssembly !== "object") {
    setStatus("This browser does not support WebAssembly.", true);
    return;
  }
  ["original", "reversed"].forEach(function (k) {
    var m = window.PAN_MODULES[k];
    if (m && m.label) (k === "original" ? el.orig : el.rev).textContent = String(m.label);
  });

  Promise.all([loadModule("original"), loadModule("reversed")]).then(function () {
    el.sound.disabled = false;
    el.orig.disabled = false;
    el.rev.disabled = false;
    el.pause.disabled = false;
    setMode("original");
    setStatus("Ready. Turn the sound on and switch between the two modules.");
    requestAnimationFrame(frame);
  }).catch(function (e) {
    setStatus("Could not load the WebAssembly modules: " + (e && e.message ? e.message : e), true);
  });
})();
</script>
