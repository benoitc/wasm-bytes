<script>
(function(){
  'use strict';

  var $ = function(id){ return document.getElementById(id); };
  var stage = $('stage'), ball = $('ball'), reading = $('reading'), marker = $('marker');
  var soundBtn = $('sound'), pauseBtn = $('pause');
  var origBtn = $('mode-original'), revBtn = $('mode-reversed');

  var PAN = window.PAN_MODULES || null;

  var modules = {};
  var currentMode = 'original';
  var soundOn = false;
  var paused = false;
  var audioCtx = null;
  var beepTimer = null;

  var TRAVERSE_MS = 6000;
  var xFloat = 0;
  var x = 0;
  var lastTs = null;

  function decodeBase64(b64){
    var clean = String(b64 || '').replace(/\s+/g, '');
    var bin = atob(clean);
    var bytes = new Uint8Array(bin.length);
    for (var i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
    return bytes;
  }

  function sha256Hex(bytes){
    if (!(window.crypto && crypto.subtle && crypto.subtle.digest)) return Promise.resolve(null);
    return crypto.subtle.digest('SHA-256', bytes).then(function(buf){
      return Array.prototype.map.call(new Uint8Array(buf), function(b){
        return b.toString(16).padStart(2, '0');
      }).join('');
    }).catch(function(){ return null; });
  }

  function screenPan(xx){
    var m = modules[currentMode];
    if (!m || !m.fn) return null;
    try {
      var v = m.fn(xx);
      var n = Number(v);
      return isNaN(n) ? null : n;
    } catch (e) {
      return null;
    }
  }

  function updateReading(){
    var W = stage.clientWidth;
    var center = 16 + (W - 32) * (x / 640);
    ball.style.left = center + 'px';

    var v = screenPan(x);
    var vText = (v === null) ? '—' : String(v);
    reading.textContent = 'x = ' + x + '\nscreen_pan(x) = ' + vText;

    var half = (reading.offsetWidth || 190) / 2;
    var minC = half + 4;
    var maxC = W - half - 4;
    var cx = center;
    if (maxC < minC) cx = W / 2;
    else cx = Math.min(maxC, Math.max(minC, cx));
    reading.style.left = cx + 'px';

    if (v === null) {
      marker.style.opacity = '0';
    } else {
      marker.style.opacity = '1';
      marker.style.left = (50 + 50 * v / 500) + '%';
    }
  }

  function tick(ts){
    if (lastTs === null) lastTs = ts;
    var dt = Math.min(100, ts - lastTs);
    lastTs = ts;
    if (!paused) {
      xFloat += dt * (640 / TRAVERSE_MS);
      if (xFloat >= 640) xFloat -= 640;
      x = Math.round(xFloat);
    }
    updateReading();
    requestAnimationFrame(tick);
  }

  function playBeep(){
    if (!soundOn || !audioCtx) return;
    if (audioCtx.state === 'suspended') {
      audioCtx.resume().catch(function(){});
    }
    var v = screenPan(x);
    if (v === null) return;
    var t = audioCtx.currentTime;
    var osc = audioCtx.createOscillator();
    var gain = audioCtx.createGain();
    osc.type = 'sine';
    osc.frequency.setValueAtTime(880, t);
    gain.gain.setValueAtTime(0.0001, t);
    gain.gain.linearRampToValueAtTime(0.22, t + 0.008);
    gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.09);
    osc.connect(gain);
    if (typeof audioCtx.createStereoPanner === 'function') {
      var pan = audioCtx.createStereoPanner();
      pan.pan.setValueAtTime(Math.max(-1, Math.min(1, v / 500)), t);
      gain.connect(pan);
      pan.connect(audioCtx.destination);
    } else {
      gain.connect(audioCtx.destination);
    }
    osc.start(t);
    osc.stop(t + 0.1);
  }

  function toggleSound(){
    if (!soundOn) {
      (async function(){
        if (navigator.audioSession) {
          try { navigator.audioSession.type = 'playback'; } catch (e) {}
        }
        try {
          if (!audioCtx) {
            var Ctor = window.AudioContext || window.webkitAudioContext;
            if (Ctor) audioCtx = new Ctor();
          }
          if (audioCtx && audioCtx.state === 'suspended') await audioCtx.resume();
        } catch (e) {}
        soundOn = true;
        soundBtn.textContent = 'Sound: on';
        soundBtn.setAttribute('aria-pressed', 'true');
        playBeep();
        if (beepTimer) clearInterval(beepTimer);
        beepTimer = setInterval(playBeep, 200);
      })();
    } else {
      soundOn = false;
      if (beepTimer) { clearInterval(beepTimer); beepTimer = null; }
      soundBtn.textContent = 'Sound: off';
      soundBtn.setAttribute('aria-pressed', 'false');
    }
  }

  function setMode(mode){
    currentMode = mode;
    origBtn.classList.toggle('selected', mode === 'original');
    revBtn.classList.toggle('selected', mode === 'reversed');
    origBtn.setAttribute('aria-pressed', String(mode === 'original'));
    revBtn.setAttribute('aria-pressed', String(mode === 'reversed'));
    updateReading();
  }

  function buildInspect(){
    var tbody = $('inspect-rows');
    tbody.textContent = '';
    if (!PAN) {
      var tr = document.createElement('tr');
      var td = document.createElement('td');
      td.colSpan = 4;
      td.textContent = 'modules.js was not found or window.PAN_MODULES is missing.';
      tr.appendChild(td);
      tbody.appendChild(tr);
      return;
    }
    ['original', 'reversed'].forEach(function(key){
      var entry = PAN[key];
      var tr = document.createElement('tr');

      var tdModule = document.createElement('td');
      tdModule.textContent = entry ? (entry.label || key) : (key + ' (missing)');

      var tdModel = document.createElement('td');
      tdModel.textContent = entry ? (entry.model == null ? '' : entry.model) : '';

      var tdReply = document.createElement('td');
      tdReply.textContent = entry ? (entry.reply == null ? '' : entry.reply) : '';

      var tdSha = document.createElement('td');

      var recLabel = document.createElement('span');
      recLabel.className = 'sub';
      recLabel.textContent = 'recorded:';
      var recCode = document.createElement('code');
      recCode.textContent = entry ? (entry.sha256 == null ? '' : entry.sha256) : '';
      tdSha.appendChild(recLabel);
      tdSha.appendChild(recCode);

      var loadLabel = document.createElement('span');
      loadLabel.className = 'sub';
      loadLabel.textContent = 'loaded:';
      var loadCode = document.createElement('code');
      loadCode.textContent = '…';
      tdSha.appendChild(loadLabel);
      tdSha.appendChild(loadCode);

      var matchLabel = document.createElement('span');
      matchLabel.className = 'sub';
      matchLabel.textContent = 'match:';
      var matchVal = document.createElement('span');
      matchVal.textContent = '…';
      tdSha.appendChild(matchLabel);
      tdSha.appendChild(matchVal);

      tr.appendChild(tdModule);
      tr.appendChild(tdModel);
      tr.appendChild(tdReply);
      tr.appendChild(tdSha);
      tbody.appendChild(tr);

      var m = modules[key];
      if (!m || !m.bytes) {
        loadCode.textContent = 'unavailable';
        matchVal.textContent = 'unavailable';
        return;
      }
      sha256Hex(m.bytes).then(function(hex){
        if (hex === null) {
          loadCode.textContent = 'unavailable';
          matchVal.textContent = 'unavailable';
        } else {
          loadCode.textContent = hex;
          var rec = entry && entry.sha256 ? String(entry.sha256).trim().toLowerCase() : '';
          matchVal.textContent = rec ? (hex.toLowerCase() === rec ? 'match' : 'mismatch') : 'no recorded value';
        }
      });
    });
  }

  function load(){
    if (!PAN) { buildInspect(); updateReading(); return; }
    Promise.all(['original', 'reversed'].map(function(key){
      var entry = PAN[key];
      if (!entry) return Promise.resolve();
      return (async function(){
        try {
          var bytes = decodeBase64(entry.bytes);
          var mod = await WebAssembly.compile(bytes);
          var instance = await WebAssembly.instantiate(mod, {});
          var fn = instance.exports && instance.exports.screen_pan;
          if (typeof fn !== 'function') throw new Error('module does not export screen_pan');
          modules[key] = { entry: entry, bytes: bytes, fn: fn };
        } catch (err) {
          modules[key] = { entry: entry, bytes: null, fn: null, error: err };
        }
      })();
    })).then(function(){
      buildInspect();
      updateReading();
    });
  }

  soundBtn.addEventListener('click', toggleSound);
  origBtn.addEventListener('click', function(){ setMode('original'); });
  revBtn.addEventListener('click', function(){ setMode('reversed'); });
  pauseBtn.addEventListener('click', function(){
    paused = !paused;
    pauseBtn.textContent = paused ? 'Play' : 'Pause';
    pauseBtn.setAttribute('aria-pressed', String(paused));
  });

  load();
  requestAnimationFrame(tick);
})();
</script>
</body>
