    tdModule.textContent = entry ? (entry.label || key) : (key + ' (missing)');
    const tdModel = document.createElement('td'); tdModel.textContent = entry ? (entry.model ?? '') : '';
    const tdReply = document.createElement('td'); tdReply.textContent = entry ? (entry.reply ?? '') : '';
    const tdSha = document.createElement('td');
    const rec = document.createElement('span'); rec.className = 'sub'; rec.textContent = 'recorded: ';
    const recCode = document.createElement('code'); recCode.textContent = entry ? (entry.sha256 ?? '') : '';
    tdSha.append(rec, recCode);
    const loadLabel = document.createElement('span'); loadLabel.className = 'sub'; loadLabel.textContent = 'loaded: ';
    const loadCode = document.createElement('code'); loadCode.textContent = '…';
    tdSha.append(loadLabel, loadCode);
    const matchLabel = document.createElement('span'); matchLabel.className='sub'; matchLabel.textContent='match: ';
    const matchVal = document.createElement('span'); matchVal.textContent = '…';
    tdSha.append(matchLabel, matchVal);
    tr.append(tdModule, tdModel, tdReply, tdSha);
    tbody.appendChild(tr);
    const m = modules[key];
    if (!m || !m.bytes) { loadCode.textContent = 'unavailable'; matchVal.textContent = 'unavailable'; continue; }
    sha256Hex(m.bytes).then(hex => {
      if (hex === null) { loadCode.textContent = 'unavailable'; matchVal.textContent = 'unavailable'; }
      else {
        loadCode.textContent = hex;
        const rec = (entry && entry.sha256 ? String(entry.sha256).trim().toLowerCase() : '');
        matchVal.textContent = rec ? (hex.toLowerCase() === rec ? 'match' : 'mismatch') : 'no recorded value';
      }
    });
  }
}
